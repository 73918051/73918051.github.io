12 } // 结束内层 for
13 if (flag) { // 如果 flag 为真（内层 5 次全部成功）

15 } else { // 否则
16 flag = 1; // 重置 flag，继续试探下一个 n
17 } // 结束 if-else
18 } // 结束外层 for
19 printf("The total number of fish is:%d \n", n); // 输出结果
