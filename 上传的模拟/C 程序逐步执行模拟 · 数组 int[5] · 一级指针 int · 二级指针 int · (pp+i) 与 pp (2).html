1#include <stdio.h> // 包含标准输入输出头文件
2int main() { // 定义主函数
3 int a[5] = {8,6,7,9,10}; // 定义数组a
4 int i, *p = a; // 一级指针p指向数组a
5 int **pp = &p; // 二级指针pp指向p的地址
6 for(i=0; i<5; i++) { // 循环输出

8 }
9 putchar('\n');
10 for(i=0; i<5; i++, p++) { // p移动
11 printf("%d ", **pp); // 每次取p指向的值
12 }
13 return 0; // 返回0，程序结束
14}
