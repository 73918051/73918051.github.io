1#include <stdio.h> // 包含标准输入输出头文件
2#include <string.h> // 包含字符串处理头文件
3int main() { // 定义主函数
4 char str[100], ch; // 定义字符串和要删除的字符
5 int i, k = 0; // 定义循环变量和计数器
6 printf("Input a string:\n"); // 提示输入字符串
7 gets(str); // 输入字符串
8 printf("Delete ?:"); // 提示输入要删除的字符
9 scanf("%c", &ch); // 输入要删除的字符
10 for(i=0; i<strlen(str); i++) { // 遍历字符串
11 if(str[i] == ch) { // 如果当前字符等于要删除的字符
12 continue; // 跳过该字符
13 }
14 str[k++] = str[i]; // 将不删除的字符保存到前面
15 }
16 str[k] = '\0'; // 在末尾添加结尾符
17 puts(str); // 输出处理后的字符串
18 return 0; // 返回0，程序结束
19}
