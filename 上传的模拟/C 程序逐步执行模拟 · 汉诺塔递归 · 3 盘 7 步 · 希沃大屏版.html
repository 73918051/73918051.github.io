1#include <stdio.h>                                        // 包含标准输入输出头文件


2void movedisk(int n, char A, char B, char C) {         // 汉诺塔递归函数（无返回值）


3    if(n == 1) {                                        // 递归出口（base case）


4        printf("%c->%c \n", A, C);                       // 直接移动，\n 换行


5    } else {                                                // 递归调用分支


6        movedisk(n-1, A, C, B);                             // 将 n-1 个从 A 经 C 移到 B


7        printf("%c->%c \t", A, C);                       // 将最大盘从 A 移到 C，\t 不换行


8        movedisk(n-1, B, A, C);                             // 将 n-1 个从 B 经 A 移到 C


9    }


10}


11int main() {                                             // 主函数


12    int n;                                              // 定义变量


13    printf("Input n:");                                     // 提示输入（末尾无 \n）


14    scanf("%d", &n);                                    // 输入 n（需要地址 &n）


15    printf("The step to move %d disks:\n", n);              // 打印标题并换行


16    movedisk(n, 'A', 'B', 'C');                             // 调用汉诺塔递归函数


17    return 0;                                           // 返回0，程序结束


18}
