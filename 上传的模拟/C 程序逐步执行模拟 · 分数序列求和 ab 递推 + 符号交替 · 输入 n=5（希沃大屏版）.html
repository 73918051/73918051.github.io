1#include <stdio.h> // 包含标准输入输出头文件
2int main() { // 定义主函数
3 float a=2, b=1, t, s=0; // a=2 分子，b=1 分母，t 未初始化，s=0 累加和
4 int sign=1, i, n; // sign=1 符号；i、n 未初始化
5 scanf("%d", &n); // 读入项数 n（演示输入 n = 5）
6 for (i=1; i<=n; i++) { // 循环 n 次，i 从 1 到 n
7 s += a/b * sign; // 累加当前项（带符号）到 s
8 printf("No.%d:%.0f/%.0f \t", i, sign*a, b); // 输出当前项（\t 制表符不换行）
9 sign = -sign; // 符号取反，实现正负交替
10 t = a; // 第 1 步：临时变量保存旧 a
11 a = a + b; // 第 2 步：新分子 = 旧分子 + 旧分母
12 b = t; // 第 3 步：新分母 = 旧分子
13 } // 结束 for 循环
14 printf("\ns=%.2f", s); // 换行后输出前 n 项之和（保留 2 位小数）
15 return 0; // 返回 0，程序正常结束
16} // main 结束，栈帧销毁
