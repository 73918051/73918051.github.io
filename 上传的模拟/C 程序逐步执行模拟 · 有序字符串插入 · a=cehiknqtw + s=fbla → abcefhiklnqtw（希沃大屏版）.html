2#include <string.h> // 包含字符串处理头文件
3int main() { // 定义主函数
4 static char a[20] = "cehiknqtw"; // 定义并初始化有序字符串
5 char s[] = "fbla"; // 定义要插入的字符串
6 int i, k, j; // 定义循环变量
7 for(k=0; k<strlen(s); k++) { // 遍历要插入的每个字符
8 j = 0; // 初始化j为0
9 while(s[k] >= a[j] && a[j] != '\0') { // 找到插入位置
10 j++;
11 }
12 for(i=strlen(a); i>=j; i--) { // 将插入位置后的元素后移
13 a[i+1] = a[i];
14 }
15 a[j] = s[k]; // 插入字符
16 }
17 puts(a); // 输出插入后的字符串
18 return 0; // 返回0，程序结束
19}
