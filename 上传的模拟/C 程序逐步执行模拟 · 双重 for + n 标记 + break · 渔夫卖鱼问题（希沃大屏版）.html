10 break; // 立即跳出内层 j 循环（只跳一层）
11 } // 结束 if-else
12 } // 结束内层 for
13 if (j == 5 && x == 11) { // 4 次出售全部完成且恰好剩 11 条？

15 n = 0; // 标记已找到，通知外层退出
16 } // 结束 if
17 } // 结束外层 for
18 return 0; // 返回 0，程序正常结束
