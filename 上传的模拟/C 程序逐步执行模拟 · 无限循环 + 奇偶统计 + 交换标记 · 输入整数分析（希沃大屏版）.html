1#include <stdio.h> // 包含标准输入输出头文件
2int main() { // 定义主函数
3 int m; // 定义输入变量（未初始化）
4 int count=0, count1=0, count2=0; // 三个计数器清零（count 含最后的 0）
5 int flag=0, temp; // flag=0（未交换）；temp 未初始化
6 printf("Please input m: \n"); // 输出提示并换行（printf 第 1 次）
7 for ( ; ; ) { // 无限循环：无终止条件，靠 break 退出
8 scanf("%d", &m); // 从输入读入一个整数到 m
9 count++; // 总数加 1（在 if 之前，0 也被计入）
10 if (m) { // m 非 0 才统计；m=0 走 else break
11 if (m % 2) { // 余 1 是奇数，余 0 是偶数
12 count1++; // 奇数个数加 1
13 } else { // 否则（偶数）
14 count2++; // 偶数个数加 1
15 } // 结束 if (m % 2)
16 } else { // m 为 0
17 break; // 立即跳出 for(;;)（唯一出口）
18 } // 结束 if (m)
19 if ((count-1) % 2) { // 除 0 外已读个数为奇数 → 交换
20 temp = count1; // 交换第 1 步：备份 count1
21 count1 = count2; // 交换第 2 步：count1 ← count2
22 count2 = temp; // 交换第 3 步：count2 ← temp
23 flag = 1; // 标记已交换
24 } // 结束 if ((count-1) % 2)
25 } // 结束 for(;;)
26 if (flag == 0) { // flag=0：未交换，走正常输出
27 printf("Total number:%d \n", count-1); // 输出总数（不含 0）
28 printf("Odd number:%d \n", count1); // 输出奇数个数
29 printf("Even number:%d \n", count2); // 输出偶数个数
30 } else { // flag=1：已交换，走"开玩笑"分支
31 printf("Hahaha...Odd:%d,Even:%d", count1, count2); // 输出（printf 第 2 次，无换行）
32 } // 结束 if-else
33 return 0; // 返回 0，程序正常结束
34} // main 结束，栈帧销毁
